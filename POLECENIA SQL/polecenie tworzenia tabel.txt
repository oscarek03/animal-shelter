-- polecenia utworzenia i wypelnienia 2 tabel, darczyncy i rejestr darowizn

CREATE TABLE darczyncy (
    id_darczyncy NUMBER (4) GENERATED ALWAYS as IDENTITY(START with 1 INCREMENT by 1) NOT NULL,
    nazwa_uzytkownika VARCHAR2(20) NOT NULL,
    imie VARCHAR2 (20), 
    nazwisko VARCHAR2 (20),
    mail VARCHAR2 (50),
    CONSTRAINT darczyncy_pk PRIMARY KEY (id_darczyncy)
);


CREATE TABLE rejestr_darowizn (
    id_darowizny NUMBER (8) GENERATED ALWAYS as IDENTITY(START with 1 INCREMENT by 1) NOT NULL,
    id_darczyncy NUMBER (4) NOT NULL,
    kwota NUMBER (10) NOT NULL,
    data_darowizny date NOT NULL,
    CONSTRAINT darowizny_pk PRIMARY KEY (id_darowizny),
    CONSTRAINT darczyncy_fk FOREIGN KEY (id_darczyncy) REFERENCES darczyncy (id_darczyncy)
);
