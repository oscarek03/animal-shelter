CREATE TABLE Adresy (
    adres_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    miasto VARCHAR2(50) NOT NULL,
    kod_pocztowy VARCHAR2(10) NOT NULL,
    ulica VARCHAR2(100) NOT NULL,
    numer_domu VARCHAR2(10) NOT NULL,
    numer_mieszkania VARCHAR2(10)
);

CREATE TABLE Zwierzaki (
    zwierze_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    imie VARCHAR2(50) NOT NULL,
    rodzaj VARCHAR2(50) NOT NULL,
    rasa VARCHAR2(50),
    wiek NUMBER,
    plec VARCHAR2(10),
    status VARCHAR2(50),
    opiekun_id NUMBER,
    kojec_id NUMBER,
    data_przyjecia DATE,
    FOREIGN KEY (kojec_id) REFERENCES Kojce(kojec_id)
);

CREATE TABLE Kojce (
    kojec_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    numer NUMBER NOT NULL,
    wielkosc VARCHAR2(50) NOT NULL
);

CREATE TABLE Pracownicy (
    pracownik_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    imie VARCHAR2(50) NOT NULL,
    nazwisko VARCHAR2(50) NOT NULL,
    pensja NUMBER,
    stanowisko VARCHAR2(50),
    adres_id NUMBER,
    data_zatrudnienia DATE,
    data_zakonczenia DATE,
    FOREIGN KEY (adres_id) REFERENCES Adresy(adres_id)
);

CREATE TABLE Rejestr_adopcji (
    adopcja_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    data_adopcji DATE NOT NULL,
    adopter_id NUMBER,
    zwierze_id NUMBER,
    pracownik_id NUMBER,
    FOREIGN KEY (adopter_id) REFERENCES Osoby_adoptujace(adopter_id),
    FOREIGN KEY (zwierze_id) REFERENCES Zwierzaki(zwierze_id),
    FOREIGN KEY (pracownik_id) REFERENCES Pracownicy(pracownik_id)
);

CREATE TABLE Osoby_adoptujace (
    adopter_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    imie VARCHAR2(50) NOT NULL,
    nazwisko VARCHAR2(50) NOT NULL,
    adres_id NUMBER,
    telefon VARCHAR2(20),
    FOREIGN KEY (adres_id) REFERENCES Adresy(adres_id)
);

CREATE TABLE Rejestr_darowizn (
    darowizna_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    darczynca_id NUMBER,
    kwota NUMBER NOT NULL,
    data DATE NOT NULL,
    FOREIGN KEY (darczynca_id) REFERENCES Osoby_darczyncy(darczynca_id)
);

CREATE TABLE Osoby_darczyncy (
    darczynca_id NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    nazwa_uzytkownika VARCHAR2(50) NOT NULL,
    imie VARCHAR2(50),
    nazwisko VARCHAR2(50),
    email VARCHAR2(100)
);

CREATE TABLE Pracownicy_Zwierzaki (
    pracownik_id NUMBER,
    zwierze_id NUMBER,
    PRIMARY KEY (pracownik_id, zwierze_id),
    FOREIGN KEY (pracownik_id) REFERENCES Pracownicy(pracownik_id),
    FOREIGN KEY (zwierze_id) REFERENCES Zwierzaki(zwierze_id)
);
