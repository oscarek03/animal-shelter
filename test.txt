        <?php
                    require_once 'db_connection.php';
                    
                    try {
                        // Przygotowanie wywołania funkcji PL/SQL
                        $sql = "
                        SELECT 
                            z.imie AS zwierzak,
                            PRACOWNIK_ZWIERZETA.ZWIERZE_ID,
                            LISTAGG(CASE WHEN p.stanowisko = 'Weterynarz' THEN p.imie || ' ' || p.nazwisko || ' (' || p.ID || ')' END, ', ') WITHIN GROUP (ORDER BY p.imie) AS weterynarz,
                            LISTAGG(CASE WHEN p.stanowisko = 'Opiekun Zwierząt' THEN p.imie || ' ' || p.nazwisko || ' (' || p.ID || ')' END, ', ') WITHIN GROUP (ORDER BY p.imie) AS opiekun,
                            LISTAGG(CASE WHEN p.stanowisko = 'Sprzątacz' THEN p.imie || ' ' || p.nazwisko || ' (' || p.ID || ')' END, ', ') WITHIN GROUP (ORDER BY p.imie) AS sprzatacz
                        FROM 
                            PRACOWNIK_ZWIERZETA
                        JOIN 
                            PRACOWNICY p ON PRACOWNIK_ZWIERZETA.PRACOWNIK_ID = p.ID
                        JOIN 
                            ZWIERZETA z ON PRACOWNIK_ZWIERZETA.ZWIERZE_ID = z.ID
                        GROUP BY 
                            z.imie, PRACOWNIK_ZWIERZETA.ZWIERZE_ID
                        ";
                        
                        $stid = oci_parse($conn, $sql);

                        // Wykonanie funkcji PL/SQL
                        oci_execute($stid);
                    
                        echo "<table>";
                        echo "<tr><th>ZWIERZAK</th><th>WETERYNARZ</th><th>OPIEKUN ZWIERZAT</th><th>SPRZATACZ</th></tr>";
                        while (($row = oci_fetch_assoc($stid)) != false) {
                            echo "<tr>";
                            echo "<td>" . htmlspecialchars($row['ZWIERZAK']) . ' (' . $row['ZWIERZE_ID']. ')' . "</td>";
                            echo "<td>" . htmlspecialchars($row['WETERYNARZ']) . "</td>";
                            echo "<td>" . htmlspecialchars($row['OPIEKUN']) . "</td>";
                            echo "<td>" . htmlspecialchars($row['SPRZATACZ']) . "</td>";
                            echo "</tr>";
                        }
                        echo "</table>";
                    
                        // Zwolnienie zasobów i zamknięcie połączenia
                        oci_free_statement($stid);
                    
                    } catch (Exception $e) {
                        echo "Wystąpił błąd: " . htmlspecialchars($e->getMessage());
                    }
        ?>
